#!/usr/bin/env python3
"""role-chain.py — run a design role chain one gated step at a time (T-2191, arc-008).

A chain file (e.g. docs/designs/agent-authorization-broker/role-chain.yaml) lists steps in order.
Each step is one framework task that plays one role (requirements collector, MVP specialist, ...).
A step may start only when the step before it is APPROVED (T-2213 slice 1, operator 2026-10-04):
  1. its independent agent review passed: "- [x] [REVIEW] Independent review: step <n> ..." is ticked. Agents
     cannot tick it; the framework's `fw reviewer verdict apply` does, only for a valid green verdict
     (D-626 + IW-7: provenance, rung, panel, digest);
  2. and, where the step `decides: true`, the operator has ticked "- [x] [REVIEW] Step <n> output approved"
     (in the active or the completed task file).
Task completion alone is NOT approval (T-2201): AEF completion does not require Human ACs, so a step
task in .tasks/completed/ without the tick is an error state that blocks the chain.

  role-chain.py status <chain.yaml>          numbered view: approved / in progress / waiting
  role-chain.py next   <chain.yaml> [--dry-run]
        exit 0  started the next step (fw work-on) or printed it (--dry-run); prints the brief
        exit 3  the current step is not approved yet (says which and what is missing)
        exit 4  the chain is finished
  role-chain.py brief  <chain.yaml> <step>   the role brief for a step (prompt, inputs, output, settles)
  role-chain.py dispatch <chain.yaml> <step> [--dry-run] [--timeout S]   (T-2214)
        start the step's role in its own TermLink session (fw termlink dispatch); refuses (exit 3) unless the
        step is the next one under the gate and no live dispatch for it exists; writes and records the brief
        (handbacks/step-NN-brief.md) and the dispatch (handbacks/step-NN-dispatch.json); exit 5 when the
        worker did not start (dispatch kick-off is fire-and-forget, T-1648)
        --findings <file>: a remediation round; the brief carries the review findings to fix
  role-chain.py round    <chain.yaml> <step> [--timeout S] [--no-review]
        one full remediation round (T-2214): findings from the newest non-green verdict of the step's review
        criterion -> dispatch -> collect -> commit (named paths only, refuses when anything else is staged)
        -> fw reviewer judge; prints the panel result. Exit 0 all green, 1 not green, 3 nothing to remediate
  role-chain.py collect  <chain.yaml> <step> [--timeout S]
        wait for the role session, then check its hand-back against the card (exit 0 valid, 1 findings)

Env: ROLE_CHAIN_ROOT overrides the project root (tests); ROLE_CHAIN_FW overrides the fw binary and
FW_DISPATCH_DIR the dispatch work directory (tests stub both).

def task_file(tid):
    for sub in ("active", "completed"):
        hits = sorted((ROOT / ".tasks" / sub).glob(f"{tid}-*.md"))
        if hits:
            return hits[0], sub
    return None, None


def task_status(tid):
    f, where = task_file(tid)
    if not f:
        return "missing"
    if where == "completed":
        return "work-completed"
    m = re.search(r"^status:\s*(\S+)", f.read_text(), re.M)
    return m.group(1) if m else "unknown"


def reviewed(step):
    """The step's independent agent review is green (ticked by `fw reviewer verdict apply`)."""
    f, _ = task_file(step["task"])
    if not f:
        return False
    pat = re.compile(rf"^\s*-\s*\[[xX]\]\s*\[REVIEW\]\s*Independent review: step {step['step']}\b", re.M)
    return bool(pat.search(f.read_text()))


LEDGER = ".context/reviews/verdicts.jsonl"
_VERDICT_RE = re.compile(r"\*\*Reviewer verdict:\*\*\s*green\s+(V-[0-9]{8}-[0-9a-f]+)")


def _sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(65536), b""):
            h.update(chunk)
    return h.hexdigest()


def review_binding(step, chain_dir):
    """T-2213 slice 2: the green review must still describe the output as it is now.
    Returns (ok, reason). Checks: (1) the review criterion names the green verdict that ticked it;
    (2) that verdict's reviewed revision is in the ledger; (3) the output is unchanged since that revision;
    (4) the hand-back record's output sha256 equals the output file now."""
    f, _ = task_file(step["task"])
    if not f:
        return False, "task missing"
    text = f.read_text()
    blk = re.search(rf"- \[([ xX])\] \[REVIEW\]\s*Independent review: step {step['step']}\b(.*?)(?=\n- \[|\n#|\Z)",
                    text, re.S)
    m = _VERDICT_RE.search(blk.group(2)) if blk else None
    how = "green"
    if not m and blk and blk.group(1).lower() == "x":
        # T-2353: the panel was not green and the operator accepted the step anyway (their tick, Watchtower
        # provenance). The task names the accepted verdict OUTSIDE the criterion (a line in Decisions), because the
        # tick's provenance digest covers the criterion text; the same bindings apply as for a green one.
        m = re.search(rf"\*\*Operator accepted:\*\*\s*step {step['step']}\s+(V-[0-9]{{8}}-[0-9a-f]+)", text)
        how = "operator-accepted"
    if not m:
        return False, ("review criterion names no green verdict" if not (blk and blk.group(1).lower() == "x")
                       else "ticked, but names neither a green verdict nor an operator-accepted one")
    vid, rev = m.group(1), None
    try:
        for line in open(ROOT / LEDGER):
            try:
                row = json.loads(line)
            except json.JSONDecodeError:
                continue
            if row.get("id") == vid:
                rev = row.get("revision")
    except OSError:
        return False, "verdict ledger unreadable"
    if not rev:
        return False, f"verdict {vid} not in the ledger"
    out = step.get("output")
    if out:
        r = subprocess.run(["git", "-C", str(ROOT), "diff", "--quiet", rev, "--", out], capture_output=True)
        if r.returncode == 1:
            return False, f"output changed after the reviewed revision {rev[:9]}; review again"
        if r.returncode not in (0, 1):
            return False, f"cannot compare output with reviewed revision {rev[:9]}"
        rec = handback_record(step, chain_dir)
        if rec is None:
            return False, "no hand-back record"
        if not (ROOT / out).is_file() or rec.get("output", {}).get("sha256") != _sha256(ROOT / out):
            return False, "hand-back output hash does not match the output file"
    return True, f"bound to {vid} @ {rev[:9]}" + ("" if how == "green" else " (operator-accepted)")


def handback_records(step, chain_dir):
    """All role-handback/1 records of a step (other step-NN-*.json files are dispatch and render records)."""
    out = []
    for f in sorted((chain_dir / "handbacks").glob(f"step-{step['step']:02d}-*.json")):
        try:
            rec = json.loads(f.read_text())
        except (OSError, json.JSONDecodeError):
            continue
        if rec.get("schema") == "role-handback/1":
            out.append(rec)
    return out


def handback_record(step, chain_dir):
    """The current revision's hand-back: the record whose task is the step's task (T-2213 revisions); a record
    without a task field counts only when it is the step's only record."""
    recs = handback_records(step, chain_dir)
    own = [r for r in recs if r.get("task") == step["task"]]
    if own:
        return own[-1]
    return recs[0] if len(recs) == 1 and not recs[0].get("task") else None


def operator_approved(step):
    f, where = task_file(step["task"])
    if not f:
        return False
    pat = re.compile(rf"^\s*-\s*\[[xX]\]\s*\[REVIEW\]\s*Step {step['step']} output approved", re.M)
    return bool(pat.search(f.read_text()))


CHAIN_DIR = None  # set in main(): directory of the chain file (hand-backs live in <dir>/handbacks)


def approved(step):
    """Agent review green AND bound to the current output AND (operator approval where the step decides)."""
    if not reviewed(step):
        return False
    if CHAIN_DIR is not None and not review_binding(step, CHAIN_DIR)[0]:
        return False
    return operator_approved(step) if step.get("decides", True) else True


def state(step):
    if approved(step):
        return "approved"
    st = task_status(step["task"])
    if st == "missing":
        return "missing task"
    if st == "work-completed":
        return "ERROR: completed without operator approval"
    if st in ("started-work", "issues"):
        out = ROOT / step["output"] if step.get("output") else None
        if out and out.exists():
            if not reviewed(step):
                return "awaiting independent review"
            ok, why = review_binding(step, CHAIN_DIR) if CHAIN_DIR is not None else (True, "")
            return "awaiting operator approval" if ok else f"review stale: {why}"
        return "in progress"
    return "waiting"


def load(chain):
    d = yaml.safe_load(Path(chain).read_text())
    steps = sorted(d["steps"], key=lambda s: s["step"])
    return d, steps


def prompt_files(d, s):
    """The files a role session receives, in order (T-2204)."""
    return [x for x in (d.get("rules"), d.get("common"), s.get("prompt"), d.get("adapter"), d.get("profile")) if x]


def brief(d, s):
    # Prompt composition order (T-2204): project rules, rules for every card, the role card, adapter, profile.
    compose = prompt_files(d, s)
    lines = [f"Step {s['step']}: {s['role']}  ({s['task']}, chain {d.get('chain')})",
             f"  {s['step']}.0 Prompt, in order: {' + '.join(compose) or '(none)'}",
             f"  {s['step']}.1 Role card: {s.get('prompt') or '(none)'}",
             f"  {s['step']}.2 Decides (operator-only review): {'yes' if s.get('decides') else 'no'}"
             + (f"; iterates with step {s['iterates_with']}" if s.get('iterates_with') else ""),
             f"  {s['step']}.3 Inputs: {', '.join(map(str, s.get('inputs') or [])) or '(none)'}",
             f"  {s['step']}.4 Output: {s.get('output')}",
             f"  {s['step']}.5 Settles: {', '.join(s.get('settles') or []) or '(nothing extra)'}",
             f"  {s['step']}.6 Operator: {s.get('operator') or '-'}",
             f"  {s['step']}.7 Hand-over: fw reviewer judge {s['task']} -> green -> fw reviewer verdict apply; "
             + (f"then the operator ticks \"[REVIEW] Step {s['step']} output approved\"; " if s.get('decides', True) else "")
             + "then: role-chain.py next"]
    return "\n".join(lines)


def input_drift(s):
    """Inputs whose sha256 changed since the step's hand-back (an approved step built on an input that moved)."""
    rec = handback_record(s, CHAIN_DIR) if CHAIN_DIR is not None else None
    moved = []
    for inp in (rec or {}).get("inputs", []):
        p = ROOT / inp.get("path", "")
        if p.is_file() and _sha256(p) != inp.get("sha256"):
            moved.append(inp["path"])
    return moved


def cmd_status(d, steps):
    print(f"Role chain {d.get('chain')} ({d.get('arc')})")
    for s in steps:
        st = state(s)
        extra = f"  (supersedes {', '.join(s['revisions'])})" if s.get("revisions") else ""
        line = f"  {s['step']}. {s['role']:<45} {s['task']}  {st}{extra}"
        if st == "approved":
            moved = input_drift(s)
            if moved:
                line += f"\n       note: inputs changed since its hand-back: {', '.join(moved)}"
        print(line)
    return 0


def cmd_next(d, steps, dry):
    for i, s in enumerate(steps):
        if approved(s):
            continue
        st = task_status(s["task"])
        if st == "missing":
            print(f"Step {s['step']} task {s['task']} does not exist.", file=sys.stderr)
            return 3
        if st == "work-completed":
            print(f"Step {s['step']} ({s['role']}, {s['task']}) is completed but lacks "
                  f"{'its green independent review' if not reviewed(s) else 'the operator approval tick'}. "
                  f"Completion is not approval; the chain stays here.", file=sys.stderr)
            return 3
        if st in ("started-work", "issues"):
            need = ("a green independent review (fw reviewer judge, then fw reviewer verdict apply)"
                    if not reviewed(s) else f"the operator's tick on \"[REVIEW] Step {s['step']} output approved\"")
            print(f"Step {s['step']} ({s['role']}, {s['task']}) is {state(s)}. The next role starts only after "
                  f"{need} in {s['task']}.")
            return 3
        # first unapproved step, not started: everything before it is approved by construction
        if dry:
            print(f"Would start step {s['step']}: {s['task']}")
        else:
            fw = ROOT / ".agentic-framework" / "bin" / "fw"
            r = subprocess.run([str(fw), "work-on", s["task"]], cwd=ROOT)
            if r.returncode != 0:
                print(f"fw work-on {s['task']} failed (exit {r.returncode})", file=sys.stderr)
                return r.returncode
        print(brief(d, s))
        return 0
    print(f"Chain {d.get('chain')} finished: all {len(steps)} steps approved.")
    return 4

ROLE_TOOLS = "Read,Write,Edit,Bash,Glob,Grep"  # no MCP, no TermLink tools (T-2200: no exec/keystroke powers)


def fw_bin():
    return os.environ.get("ROLE_CHAIN_FW") or str(ROOT / ".agentic-framework" / "bin" / "fw")


def dispatch_dir():
    return Path(os.environ.get("FW_DISPATCH_DIR", "/tmp/tl-dispatch"))


def role_slug(s):
    return Path(s["prompt"]).stem if s.get("prompt") else re.sub(r"[^a-z0-9]+", "-", s["role"].lower()).strip("-")


def session_name(s, round_no=1):
    base = f"role-{s['step']:02d}-{role_slug(s)}"
    return base if round_no <= 1 else f"{base}-r{round_no}"


def next_round(s):
    """First round number whose session name has no dispatch directory yet (each round gets its own session)."""
    k = 1
    while (dispatch_dir() / session_name(s, k)).exists():
        k += 1
    return k


def last_dispatched_session(s):
    rec = CHAIN_DIR / "handbacks" / f"step-{s['step']:02d}-dispatch.json" if CHAIN_DIR is not None else None
    try:
        return json.loads(rec.read_text()).get("session") or session_name(s)
    except (OSError, json.JSONDecodeError, AttributeError):
        return session_name(s)


def rel(p):
    p = Path(p)
    return str(p.relative_to(ROOT)) if p.is_absolute() and p.is_relative_to(ROOT) else str(p)


def worker_live(name):
    w = dispatch_dir() / name
    if not w.is_dir() or (w / "exit_code").exists():
        return False
    return subprocess.run(["pgrep", "-f", "--", f"{w}/run.sh"], capture_output=True).returncode == 0


def next_unapproved(steps):
    for s in steps:
        if not approved(s):
            return s
    return None


def compose_brief(d, s, chain_dir, findings=None):
    """The full text a role session receives: prompt files in chain order, then the step's facts."""
    nn = f"{s['step']:02d}"
    hb = rel(chain_dir / "handbacks" / f"step-{nn}-{role_slug(s)}.json")
    rc = rel(chain_dir / "handbacks" / f"step-{nn}-render-check.json")
    parts = [f"# Role session: step {s['step']} — {s['role']} ({s['task']}, chain {d.get('chain')})",
             "",
             "You are the role below and nothing else. The orchestrator (ring20) dispatched you; it reviews your "
             "output and asks the operator. Read every section; the step facts at the end are binding.", ""]
    for i, f in enumerate(prompt_files(d, s), 1):
        parts += [f"## Part {i}: `{f}`", "", (ROOT / f).read_text().rstrip(), ""]
    tf, _ = task_file(s["task"])
    facts = [f"F.1 Task: {s['task']}. Start with `.agentic-framework/bin/fw work-on {s['task']}`, then read the task file "
             f"`{rel(tf) if tf else s['task']}` (its Context and Acceptance Criteria are binding).",
             f"F.2 Inputs (read all; their sha256 now, record them in the hand-back):"]
    for j, inp in enumerate(s.get("inputs") or [], 1):
        p = ROOT / inp
        if inp == s.get("output"):
            continue
        facts.append(f"  F.2.{j} `{inp}` sha256 {_sha256(p) if p.is_file() else '(missing)'}")
    facts += [f"F.3 Output: `{s.get('output')}`. Settles: {', '.join(s.get('settles') or []) or '-'}.",
              f"F.4 Hand-back record (role-handback/1): `{hb}`; check it with "
              f"`python3 scripts/role-handback-check.py {hb} --card {s.get('prompt')}` until it says VALID.",
              f"F.5 Render check for the drawings: `python3 scripts/design-render-check.py {s.get('output')} --out {rc}`; "
              f"name it in the hand-back as render_check.",
              f"F.6 Commit after each unit of work, naming paths only: `git commit -m \"{s['task']}: ...\" -- <paths>` "
              f"(never `git add -A`; check `git diff --cached --name-only` first; other sessions share the index).",
              "F.7 You must not: run `fw reviewer` or any review command; tick any acceptance criterion; publish, share or "
              "upload anything outside the estate; send content to external models; use TermLink exec, inject or "
              "remote commands; read or answer the project's sidecar inbox or any peer message (they belong to the "
              "project's main session); change files outside your output, your hand-back and your render check.",
              f"F.8 Questions only the operator can answer: write them in the output as open questions and in the "
              f"hand-back (open_questions); do not wait for an answer.",
              f"F.9 When the hand-back is VALID and committed, print one last line `HANDBACK {hb}` and stop."]
    crs = []
    for later in d.get("steps", []):
        if later["step"] <= s["step"] or not approved(later):
            continue
        rec = handback_record(later, chain_dir) or {}
        crs += [f"step {later['step']} ({later['task']}): {c.get('what')}" for c in rec.get("change_requests", [])
                if c.get("step") == s["step"]]
    if s.get("revisions"):
        facts.insert(1, f"F.1a This is a revision; it supersedes {', '.join(s['revisions'])}. Change only what the task's "
                        f"Context names; keep every existing id and number stable.")
    if crs:
        facts.append("F.10 Change requests to apply (raised by approved later steps; the operator's decision is in "
                     "the raising step's task):")
        facts += [f"  F.10.{k} {c}" for k, c in enumerate(crs, 1)]
    if findings:
        facts.append(f"F.11 This is a remediation round. The independent review found the items below (evidence: "
                     f"`{rel(findings)}`). Fix each one in the output, keep everything else as it is, add a version history "
                     f"row, re-run the render check and refresh the hand-back.")
        facts += ["  " + line for line in Path(findings).read_text().splitlines() if line.strip()]
    parts += ["## Step facts (binding)", ""] + facts
    return "\n".join(parts) + "\n"


def cmd_dispatch(d, steps, n, dry, timeout, findings=None):
    match = [s for s in steps if s["step"] == n]
    if not match:
        print(f"no step {n}", file=sys.stderr)
        return 2
    s = match[0]
    nxt = next_unapproved(steps)
    if nxt is None or nxt["step"] != n:
        print(f"Step {n} is not the next step (next: {nxt['step'] if nxt else 'none, chain finished'}); "
              f"a role starts only after the step before it is approved.", file=sys.stderr)
        return 3
    prev = last_dispatched_session(s)
    if worker_live(prev) or worker_live(session_name(s)):
        print(f"A live dispatch for step {n} exists ({prev}); collect it or let it finish first.", file=sys.stderr)
        return 3
    round_no = next_round(s)
    name = session_name(s, round_no)
    hbdir = CHAIN_DIR / "handbacks"
    nn = f"{n:02d}"
    brief_path = hbdir / f"step-{nn}-brief.md"
    text = compose_brief(d, s, CHAIN_DIR, findings)
    cmd = [fw_bin(), "termlink", "dispatch", "--task", s["task"], "--name", name, "--prompt-file", str(brief_path),
           "--project", str(ROOT), "--task-type", "build", "--tools", ROLE_TOOLS, "--permission-mode", "acceptEdits",
           "--strict-mcp-config", "--timeout", str(timeout)]
    if dry:
        print(f"Would write {rel(brief_path)} ({len(text)} bytes) and run:\n  " + " ".join(cmd))
        return 0
    hbdir.mkdir(parents=True, exist_ok=True)
    brief_path.write_text(text)
    r = subprocess.run(cmd, cwd=ROOT)
    if r.returncode != 0:
        print(f"fw termlink dispatch failed (exit {r.returncode})", file=sys.stderr)
        return r.returncode
    w = dispatch_dir() / name
    started = False
    for _ in range(int(os.environ.get("ROLE_CHAIN_START_WAIT", "30"))):
        if (w / "exit_code").exists() or worker_live(name):
            started = True
            break
        time.sleep(1)
    if not started:
        print(f"Dispatch of {name} returned but no worker is running in {w} (fire-and-forget kick-off, T-1648). "
              f"Recover: termlink inject {name} \"bash {w}/run.sh ...\" --enter, or dispatch again.", file=sys.stderr)
        return 5
    rec = {"schema": "role-dispatch/1", "chain": d.get("chain"), "step": n, "task": s["task"], "session": name,
           "round": round_no, "findings": rel(findings) if findings else None,
           "worker_kind": "claude", "brief": rel(brief_path), "brief_sha256": _sha256(brief_path),
           "dispatch_dir": str(w), "started": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())}
    (hbdir / f"step-{nn}-dispatch.json").write_text(json.dumps(rec, indent=2) + "\n")
    print(f"Started {name} for step {n} ({s['task']}). Brief {rel(brief_path)}; record step-{nn}-dispatch.json. "
          f"Commit both, then: role-chain.py collect <chain> {n}")
    return 0


def bind_criterion(step, out_rel, out_sha):
    """Name the exact output under review in the step's review criterion (T-2213). A changed line changes the
    criterion's digest, so the framework's `fw reviewer verdict apply` withdraws a tick whose verdict covered an
    earlier output, and the judge reviews again. Returns True when the line changed."""
    f, where = task_file(step["task"])
    if not f or where != "active":
        return False
    lines = f.read_text().split("\n")
    head = re.compile(rf"^\s*-\s*\[[ xX]\]\s*\[REVIEW\]\s*Independent review: step {step['step']}\b")
    idx = next((i for i, l in enumerate(lines) if head.search(l)), None)
    if idx is None:
        return False
    want = f"  **Output under review:** `{out_rel}` sha256 {out_sha}"
    j = idx + 1
    while j < len(lines) and lines[j].startswith("  ") and not lines[j].lstrip().startswith("- ["):
        if lines[j].startswith("  **Output under review:**"):
            if lines[j] == want:
                return False
            lines[j] = want
            f.write_text("\n".join(lines))
            return True
        j += 1
    lines.insert(idx + 1, want)
    f.write_text("\n".join(lines))
    return True


def cmd_collect(d, steps, n, timeout):
    match = [s for s in steps if s["step"] == n]
    if not match:
        print(f"no step {n}", file=sys.stderr)
        return 2
    s = match[0]
    name = last_dispatched_session(s)
    subprocess.run([fw_bin(), "termlink", "wait", "--name", name, "--timeout", str(timeout)], cwd=ROOT)
    if worker_live(name):
        print(f"{name} is still running after {timeout}s; collect again later.")
        return 3
    hb = CHAIN_DIR / "handbacks" / f"step-{n:02d}-{role_slug(s)}.json"
    if not hb.is_file():
        print(f"{name} finished without a hand-back record at {rel(hb)}; read {dispatch_dir() / name}/result.md "
              f"and dispatch again with the findings.", file=sys.stderr)
        return 1
    r = subprocess.run([sys.executable, str(ROOT / "scripts" / "role-handback-check.py"), str(hb), "--root", str(ROOT),
                        "--card", s["prompt"]], capture_output=True, text=True)
    print(r.stdout.rstrip())
    if r.returncode != 0:
        print(f"Send these findings back to the role (dispatch step {n} again with them).", file=sys.stderr)
        return 1
    if s.get("output") and (ROOT / s["output"]).is_file():
        if bind_criterion(s, s["output"], _sha256(ROOT / s["output"])):
            subprocess.run([fw_bin(), "reviewer", "verdict", "apply", s["task"]], cwd=ROOT, capture_output=True)
            print(f"Review criterion of {s['task']} now names this output; a tick for an earlier output is withdrawn "
                  f"by fw reviewer verdict apply. Commit the task file before the review.")
    if s.get("revisions") and s.get("output"):
        base = baseline_revision(s)
        if base:
            diff = subprocess.run(["git", "-C", str(ROOT), "diff", base, "--", s["output"]], capture_output=True, text=True).stdout
            dpath = CHAIN_DIR / "handbacks" / f"step-{n:02d}-baseline.diff"
            dpath.write_text(f"# Baseline for review: {s['output']} at {base[:12]} (last hand-back of {s['revisions'][-1]}) "
                             f"versus the working tree now. Every change below must be inside the revision task's scope.\n"
                             + diff)
            print(f"Baseline diff for the reviewers: {rel(dpath)} (against {base[:12]}). Commit it before the review.")
    print(f"Next (orchestrator): .agentic-framework/bin/fw reviewer judge {s['task']} --criterion 1  (detached)")
    return 0


def baseline_revision(s):
    """The commit of the superseded revision's last hand-back record: the output as it was approved before.

    T-2358: the new revision's role overwrites the same hand-back file in place, so after `collect` the working tree
    no longer holds a record for the superseded task. Search the files' git history (newest first) for the last
    commit whose version of the record names that task; the working-tree check stays as the fast path."""
    prev = s["revisions"][-1]
    files = sorted((CHAIN_DIR / "handbacks").glob(f"step-{s['step']:02d}-*.json"))
    for f in files:
        try:
            if json.loads(f.read_text()).get("task") != prev:
                continue
        except (OSError, json.JSONDecodeError):
            continue
        r = subprocess.run(["git", "-C", str(ROOT), "log", "-1", "--format=%H", "--", str(f)], capture_output=True, text=True)
        return r.stdout.strip() or None
    # History search: only the hand-back RECORD (step-NN-<role>.json). step-NN-dispatch.json and the brief carry the
    # same task id but are written at dispatch time, before the output exists, so they would pick a wrong baseline.
    for f in [CHAIN_DIR / "handbacks" / f"step-{s['step']:02d}-{role_slug(s)}.json"]:
        relf = str(f.relative_to(ROOT))
        log = subprocess.run(["git", "-C", str(ROOT), "log", "--format=%H", "--", relf], capture_output=True, text=True)
        for h in log.stdout.split():
            shown = subprocess.run(["git", "-C", str(ROOT), "show", f"{h}:{relf}"], capture_output=True, text=True)
            try:
                if shown.returncode == 0 and json.loads(shown.stdout).get("task") == prev:
                    return h
            except json.JSONDecodeError:
                continue
    return None


def latest_findings(s):
    """The newest non-green verdict for the step's review criterion (AC 1): (evidence path, judgement dict)."""
    led = ROOT / LEDGER
    rows = []
    try:
        for line in open(led):
            try:
                r = json.loads(line)
            except json.JSONDecodeError:
                continue
            if r.get("task") == s["task"] and r.get("ac") == 1:
                rows.append(r)
    except OSError:
        return None, None
    if not rows or rows[-1].get("outcome") == "green":
        return None, None
    last = rows[-1]
    ev = next((e for e in (last.get("evidence") or []) if str(e).endswith(".md")), None)
    if not ev or not (ROOT / ev).is_file():
        return ev, {"findings": [{"text": last.get("guidance") or "see evidence", "severity": "med"}], "conditions": []}
    m = re.search(r"```reviewer-judgement\s*(\{.*?\})\s*```", (ROOT / ev).read_text(), re.S)
    try:
        return ev, json.loads(m.group(1)) if m else {"findings": [{"text": last.get("guidance") or "see evidence",
                                                                     "severity": "med"}], "conditions": []}
    except json.JSONDecodeError:
        return ev, {"findings": [{"text": last.get("guidance") or "see evidence", "severity": "med"}], "conditions": []}


def _git(*args):
    return subprocess.run(["git", "-C", str(ROOT), *args], capture_output=True, text=True)


def _commit(paths, msg):
    if _git("diff", "--cached", "--name-only").stdout.strip():
        print("Refusing to commit: other files are staged in the shared index.", file=sys.stderr)
        return False
    paths = [str(p) for p in paths if (ROOT / p).exists() and _git("status", "--porcelain", "--", str(p)).stdout.strip()]
    if not paths:
        return True
    _git("add", "--", *paths)
    r = _git("commit", "-q", "-m", msg + "\n\nCo-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>", "--", *paths)
    return r.returncode == 0


def cmd_round(d, steps, n, timeout, review=True):
    match = [s for s in steps if s["step"] == n]
    if not match:
        print(f"no step {n}", file=sys.stderr)
        return 2
    s = match[0]
    ev, j = latest_findings(s)
    if j is None:
        print(f"Step {n} ({s['task']}): no non-green verdict to remediate.")
        return 3
    hb = CHAIN_DIR / "handbacks"
    k = 1
    while (hb / f"step-{n:02d}-findings-r{k}.md").exists():
        k += 1
    fpath = hb / f"step-{n:02d}-findings-r{k}.md"
    out = [f"Review findings for {s['task']} (evidence {ev}). Fix each one; keep everything else; add a version history "
           f"row; re-run the render check; refresh the hand-back."]
    out += [f"{i}. ({f.get('severity', '?')}) {f.get('text')}" for i, f in enumerate(j.get("findings", []), 1)]
    if j.get("conditions"):
        out.append("Conditions the reviewer will verify:")
        out += [f"C{i}. {c.get('text')} — verify: {c.get('verify')}" for i, c in enumerate(j["conditions"], 1)]
    fpath.write_text("\n".join(out) + "\n")
    if not _commit([rel(fpath)], f"{s['task']}: review findings r{k} for the role session"):
        return 1
    rc = cmd_dispatch(d, steps, n, False, timeout, fpath)
    if rc != 0:
        return rc
    if not _commit([rel(hb / f"step-{n:02d}-brief.md"), rel(hb / f"step-{n:02d}-dispatch.json")],
                   f"{s['task']}: remediation round dispatched ({last_dispatched_session(s)})"):
        return 1
    rc = cmd_collect(d, steps, n, timeout)
    if rc != 0:
        return rc
    tf, _ = task_file(s["task"])
    if not _commit([rel(tf), rel(hb / f"step-{n:02d}-baseline.diff")],
                   f"{s['task']}: remediation round collected (VALID); criterion bound to the new output"):
        return 1
    if not review:
        return 0
    r = subprocess.run([fw_bin(), "reviewer", "judge", s["task"], "--criterion", "1"], cwd=ROOT,
                       capture_output=True, text=True)
    lines = [l for l in (r.stdout + r.stderr).splitlines() if "seat" in l or "AC#1" in l]
    print("\n".join(lines[-6:]))
    _, after = latest_findings(s)
    return 0 if after is None else 1


def main(argv=None):
    ap = argparse.ArgumentParser(description=__doc__.splitlines()[0])
    ap.add_argument("cmd", choices=["status", "next", "brief", "dispatch", "collect", "round"])
    ap.add_argument("chain")
    ap.add_argument("step", nargs="?", type=int)
    ap.add_argument("--dry-run", action="store_true")
    ap.add_argument("--timeout", type=int, default=3600)
    ap.add_argument("--findings", help="dispatch: a file of review findings for a remediation round")
    ap.add_argument("--no-review", action="store_true", help="round: stop after collect")
    a = ap.parse_args(argv)
    global CHAIN_DIR
    CHAIN_DIR = Path(a.chain).resolve().parent
    d, steps = load(a.chain)
    if a.cmd == "status":
        return cmd_status(d, steps)
    if a.cmd == "next":
        return cmd_next(d, steps, a.dry_run)
    if a.cmd == "round":
        if a.step is None:
            print("round needs a step number", file=sys.stderr)
            return 2
        if a.dry_run:
            # round dispatches, commits and judges; it has no dry mode. Refuse rather than run for real.
            print("round has no --dry-run; use: dispatch <chain> <step> --dry-run", file=sys.stderr)
            return 2
        return cmd_round(d, steps, a.step, a.timeout, not a.no_review)
    if a.cmd in ("dispatch", "collect") and a.step is None:
        print(f"{a.cmd} needs a step number", file=sys.stderr)
        return 2
    if a.cmd == "dispatch":
        return cmd_dispatch(d, steps, a.step, a.dry_run, a.timeout, Path(a.findings) if a.findings else None)
    if a.cmd == "collect":
        return cmd_collect(d, steps, a.step, a.timeout)
    match = [s for s in steps if s["step"] == a.step]
    if not match:
        print(f"no step {a.step}", file=sys.stderr)
        return 2
    print(brief(d, match[0]))
    return 0


if __name__ == "__main__":
    sys.exit(main())
