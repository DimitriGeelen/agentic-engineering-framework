"""role-handback-check.py — validate a role hand-back record (role-handback/1, T-2212, arc-008).

  role-handback-check.py <record.json> [--card <card.md>] [--root <dir>]

import argparse
import hashlib
import json
import re
import sys
from pathlib import Path

DEFAULT_ROOT = Path(__file__).resolve().parent.parent
SCHEMA = Path("docs/designs/agent-authorization-broker/schemas/role-handback-1.schema.json")


def sha256(path: Path) -> str:
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(65536), b""):
            h.update(chunk)
    return h.hexdigest()


def card_conditions(card: Path) -> int:
    """Count the numbered conditions under the card's '## N Completion conditions' heading."""
    text = card.read_text()
    m = re.search(r"^##\s+\d*\s*Completion conditions\s*\n(.*?)(?=^##\s|\Z)", text, re.M | re.S)
    if not m:
        return -1
    return len(re.findall(r"^\d+\.\d+\s", m.group(1), re.M))


def card_drawings(card: Path) -> list[str]:
    """The required drawing ids (D-1, D-2, ...) listed in the card's 'Required drawings' section."""
    m = re.search(r"^###\s+[\d.]*\s*\S*\s*Required drawings.*?\n(.*?)(?=^##|\Z)", card.read_text(), re.M | re.S)
    return re.findall(r"\*\*(D-\d+)\*\*", m.group(1)) if m else []


def output_drawings(text: str) -> tuple[int, set[str]]:
    """Count Mermaid blocks and the drawing ids captioned in the output."""
    blocks = len(re.findall(r"^```mermaid\s*$", text, re.M))
    return blocks, set(re.findall(r"Drawing (D-\d+)\b", text))
